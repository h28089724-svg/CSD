        function renderAttachmentNav() {
            const grid = document.getElementById('attachmentNavGrid');
            if (!grid || grid.dataset.rendered === 'true') return;

            grid.innerHTML = attachmentNavItems.map((item, index) => `
                <button class="attachment-nav-btn" type="button"
                        title="${item.name}"
                        aria-label="${item.name}"
                        onclick="selectAttachmentNav(this, '${item.url}')">
                    <img src="../../images/${item.image}" alt="${item.name}">
                </button>
            `).join('');

            grid.dataset.rendered = 'true';
        }

        function closeAttachmentNav() {
            const wrap = document.getElementById('attachmentNavWrap');
            if (wrap) wrap.classList.remove('open');
            document.body.classList.remove('attachment-nav-open');
        }

        function toggleAttachmentNav(el) {
            document.querySelectorAll('.mode-btn, .sub-search-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            el.classList.add('active');

            renderAttachmentNav();

            const wrap = document.getElementById('attachmentNavWrap');
            if (!wrap) return;

            const willOpen = !wrap.classList.contains('open');
            wrap.classList.toggle('open', willOpen);
            document.body.classList.toggle('attachment-nav-open', willOpen);
        }

        function selectAttachmentNav(el, url) {
            document.querySelectorAll('.attachment-nav-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            el.classList.add('active');

            /* [이번 수정] 현재는 선택 강조만 처리합니다.
               나중에 url 값을 지정하면 아래 코드로 해당 검색 페이지 연결이 가능합니다. */
            if (url) {
                setTimeout(() => {
                    window.location.href = url;
                }, 120);
            }
        }


        function toggleSubMenu(el) {
            const currentItem = el.closest('.mode-item');
            closeAttachmentNav();

            document.querySelectorAll('.mode-item').forEach(item => {
                if (item !== currentItem) {
                    item.classList.remove('open');
                }
            });

            currentItem.classList.toggle('open');
        }

        function selectAndGo(el, url) {
            closeAttachmentNav();
            document.querySelectorAll('.mode-btn, .sub-search-btn').forEach(btn => {
                btn.classList.remove('active');
            });
            el.classList.add('active');

            setTimeout(() => {
                window.location.href = url;
            }, 150);
        }

        function goHome() {
            window.location.href = '../../index.html';
        }
