        const ACTIVE_KEY = 'selected_attachment_chain_type';

        function renderAttachmentNav() {
            const grid = document.getElementById('attachmentNavGrid');
            const savedName = localStorage.getItem(ACTIVE_KEY);

            grid.innerHTML = attachmentNavItems.map(item => `
                <button class="attachment-nav-btn ${savedName === item.name ? 'active' : ''}"
                        type="button"
                        title="${item.name}"
                        aria-label="${item.name}"
                        onclick="selectAttachmentNav(this, '${item.name}', '${item.url}')">
                    <!-- [이번 수정] chain-sub 폴더 기준 이미지 경로: ../../../images/파일명 -->
                    <img src="../../../images/${item.image}" alt="${item.name}">
                </button>
            `).join('');
        }

        function selectAttachmentNav(el, name, url) {
            document.querySelectorAll('.attachment-nav-btn').forEach(btn => {
                btn.classList.remove('active');
            });

            el.classList.add('active');
            localStorage.setItem(ACTIVE_KEY, name);

            /* [이번 수정] 검색 페이지 연결 준비
               지금은 url이 비어 있으므로 이동하지 않습니다.
               나중에 attachmentNavItems의 url 값에 경로를 넣으면 자동 이동합니다. */
            if (url) {
                setTimeout(() => {
                    window.location.href = url;
                }, 120);
            }
        }

        /* [기존 유지] 홈 버튼은 체인 메인페이지로 이동
           현재 파일 위치 기준: C:/CSD/search_html/chain/chain-sub/chain_sub-attach.html
           메인 위치 기준: C:/CSD/search_html/chain/chain_main.html */
        function goHome() {
            window.location.href = '../chain_main.html';
        }

        /* [이번 수정] MYCAD 클릭 시 C:/CSD/index.html로 이동 */
        function goIndex() {
            window.location.href = '../../../index.html';
        }

        function handleBrandKey(event) {
            if (event.key === 'Enter' || event.key === ' ') {
                event.preventDefault();
                goIndex();
            }
        }

        window.onload = renderAttachmentNav;
